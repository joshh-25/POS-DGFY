**routes.js Structure:**
```javascript
import express from 'express';
import { buildAccountController } from './controllers/accountController.js';

export function createAccountRoutes(useCases) {
  const router = express.Router();
  const controller = buildAccountController(useCases);

  router.post('/register', (req, res, next) =>
    controller.register(req, res).catch(next)
  );
  router.post('/login', (req, res, next) =>
    controller.login(req, res).catch(next)
  );
  router.get('/me', authenticate, (req, res, next) =>
    controller.getMe(req, res).catch(next)
  );
  router.patch('/me', authenticate, (req, res, next) =>
    controller.updateProfile(req, res).catch(next)
  );
  router.get('/:id', authenticate, (req, res, next) =>
    controller.getAccount(req, res).catch(next)
  );

  return router;
}
```

Note: `authenticate` middleware is existing (from backend auth setup); binds req.user.id and req.user.role.

**accountController.js Structure:**
```javascript
import { sendUseCaseResult } from '../../shared/controllers/useCaseResponder.js';

export function buildAccountController(useCases) {
  return {
    async register(req, res) {
      const { email, password, first_name, last_name, phone } = req.body;
      const result = await useCases.registerAccount({
        email, password, first_name, last_name, phone
      });
      return sendUseCaseResult(res, result, 201); // 201 on success
    },

    async login(req, res) {
      const { email, password } = req.body;
      const result = await useCases.loginAccount({ email, password });
      return sendUseCaseResult(res, result, 200);
    },

    async getMe(req, res) {
      const accountId = req.user.id;
      const result = await useCases.getAccount({ accountId });
      return sendUseCaseResult(res, result, 200);
    },

    async updateProfile(req, res) {
      const accountId = req.user.id;
      const { email, password, first_name, last_name, phone } = req.body;
      const result = await useCases.updateAccountProfile({
        accountId,
        updates: { email, password, first_name, last_name, phone }
      });
      return sendUseCaseResult(res, result, 200);
    },

    async getAccount(req, res) {
      const { id: targetAccountId } = req.params;
      const requestingAccountId = req.user.id;
      const requestingRole = req.user.role;
      const result = await useCases.getAccountForAuthorization({
        accountId: targetAccountId,
        requestingAccountId,
        role: requestingRole
      });
      return sendUseCaseResult(res, result, 200);
    }
  };
}
```

**Key Requirements:**
1. Controllers DO NOT import models directly (use repository through use cases)
2. Controllers DO NOT import repositories directly (use cases own that)
3. Controllers receive use cases via dependency injection (passed in, not imported)
4. All responses use sendUseCaseResult() — consistent formatting
5. Error handling via try/catch or Promise rejection (next middleware)
6. Request body validation happens in use cases, not controllers (controllers just parse)
7. HTTP status codes: 201 for create, 200 for success, 400 for validation, 401 for auth, 403 for forbidden, 409 for conflict

**Notes on HTTP Status Codes (per ApplicationResult):**
- ApplicationResult.success → HTTP 200 (or 201 if POST create)
- ApplicationResult.failure with code 'DUPLICATE_*' → HTTP 409
- ApplicationResult.failure with code 'VALIDATION_ERROR' → HTTP 400
- ApplicationResult.failure with code 'INVALID_CREDENTIALS' → HTTP 401
- ApplicationResult.failure with code 'UNAUTHORIZED' → HTTP 403
- ApplicationResult.failure (generic) → HTTP 500 (server error)

sendUseCaseResult should map these; if it doesn't, add logic to accountController to handle mapping.

**Registration Endpoint (POST /accounts/register):**
- success: valid {email, password, first_name, last_name, phone} → HTTP 201 + account in response
- duplicate email: register twice with same email → second returns HTTP 409
- missing email: {password, first_name} → HTTP 400 (missing email)
- invalid password: password <8 chars → HTTP 400 (validation error)
- response body: {account: {id, email, first_name, ..., created_at}}

**Login Endpoint (POST /accounts/login):**
- success: {email, password} with existing account → HTTP 200 + {sessionToken, account, businesses: []}
- wrong password: correct email, wrong password → HTTP 401
- non-existent email: email not in DB → HTTP 401
- response includes: sessionToken (JWT or opaque), account, businesses array (empty in Wave 2)

**Get Me Endpoint (GET /accounts/me):**
- success: authenticated request → HTTP 200 + {account: {...}}
- unauthenticated: no session token → HTTP 401

**Update Profile Endpoint (PATCH /accounts/me):**
- success: authenticated, valid updates {email, first_name} → HTTP 200 + {account: {...}}
- duplicate email: update to email that exists on another account → HTTP 409
- unauthenticated: no session token → HTTP 401
- invalid password: update password to <8 chars → HTTP 400

**Get Account Endpoint (GET /accounts/:id):**
- self access: get own account → HTTP 200 + {account}
- admin access: admin role gets other account → HTTP 200 + {account}
- unauthorized: non-admin gets other account → HTTP 403
- not found: ID doesn't exist → HTTP 404

**Response Format (via useCaseResponder):**
- All responses have consistent structure: {success: true/false, data: {...}, error: {...}}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP Request → Controller | Untrusted input from client (JSON body, params, headers) |
| Controller → Use Case | Use cases receive parsed input; controller doesn't validate |
| Use Case → Repository | Use cases enforce business rules; repository trusts use cases |
| Session Token → Request | JWT/opaque token from client; middleware verifies authenticity |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-04-06 | Information Disclosure | Account Endpoint | medium | mitigate | GET /accounts/:id enforces access control (self/admin only) via getAccountForAuthorization use case; non-admin returns HTTP 403 |
| T-04-07 | Spoofing | Session Token | high | mitigate | Session token generated by backend auth (per D-07 existing pattern); client must provide token in Authorization header; middleware verifies before reaching controller |
| T-04-08 | Denial of Service | Registration Endpoint | medium | accept | Rate limiting is not in scope for Wave 2 (deferred to infrastructure layer or Phase 5); controller accepts all requests |
| T-04-09 | Tampering | Request Body | medium | mitigate | Controller parses JSON; use case validates all fields (email format, password length, etc.); invalid input returns HTTP 400 |
| T-04-SC | Supply Chain | npm/pip installs | high | mitigate | No new dependencies in Wave 2 (bcryptjs, express, sequelize already in package.json); slopcheck + blocking checkpoint if new deps added |

</threat_model>
